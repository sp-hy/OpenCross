# PS1 project symbol list: battle init overlay btlinit.prg (base 0x801EE000).
# Format: name = 0xADDR;  # function|data: description. Names come from this project's own reverse engineering.
# Apply with tools\ghidra\ApplySymbols.java <this file> (see symbols\README.md).
sys_ngplus_is_carry_id = 0x801EE05C;  # function: Tests whether an id is carried (below 0x7E, outside the 12 summons).
sys_ngplus_carry_over = 0x801EE0B0;  # function: Play-init that ORs cleared bitmaps into the New Game+ carry-over words.
btl_init_display_setup = 0x801EE140;  # function: Sets up the battle display and draw environment (0,10,256,216).
btl_init_display_setup_e178 = 0x801EE178;  # function: Sets up the battle display: 320x224 buffer with GTE offset (160,112) and screen distance 512.
btl_open_battle_prd = 0x801EE364;  # function: Opens the battle.prd archive during battle init.
btl_load_stage_file = 0x801EE5A8;  # function: Loads the stage background file chosen by the battlefield id and flags the fight as in progress.
btl_unit_refresh_stat_totals = 0x801EE850;  # function: Rebuilds a unit weapon, armour and skip-mask totals from its equipment.
btl_unit_copy_equipment = 0x801EED60;  # function: Copies a character weapon, armour and accessory stat records into the battle unit.
btl_player_spawn = 0x801EEE4C;  # function: Spawns a party member unit in battle and runs its start-of-battle script; in a monster-party scene the slot takes the picked monster record instead of the character.
btl_enemy_size_model = 0x801EF7CC;  # function: Sizes and prepares an enemy model and its weapon at spawn.
btl_enemy_spawn = 0x801EFB6C;  # function: Spawns one enemy unit: applies spawn script, model and weapon setup.
btl_spawn_enemies = 0x801F07D8;  # function: Loops over the scene enemies and spawns each one.
btl_unit_build_grid = 0x801F0FD8;  # function: Builds a unit's battle grid from its record cells (only cells without the 0x4000 lock bit), the cell flags, and the gauge cap (deepest row with an element).
btl_unit_reset_gauge_and_grid = 0x801F13F4;  # function: Sets a unit's gauge to -1 and builds its battle grid.
btl_enemy_spawn_copy_record = 0x801F1434;  # function: Copies a monster record into a unit and doubles HP in New Game Plus; also used for the party slots of a monster-party scene.
btl_script_build_contexts = 0x801F19B4;  # function: Builds one script interpreter context per battle entity; script variables 0x00..0x0E and 0x60..0x7E start as the field variables 0x3D0..0x3DE and 0x3E0..0x3FE.
btl_scene_select = 0x801F1F18;  # function: Maps the battle id to a scene record and reads the scene entry; overwrites the battle id word with the scene number, so every later test compares the scene.
btl_init_main = 0x801F2A48;  # function: Battle init driver: audio, table loads, scene select, enemy spawn, unit arrays and party setup.
btl_start_transition_sequence = 0x801F2AFC;  # function: Starts the embedded transition sequence at battle start.
btl_init_clear_screen_white = 0x801F2BEC;  # function: Copies the displayed frame to VRAM and clears the screen to white.
btl_transition_sequence_blob = 0x801F2D14;  # data: Embedded transition sequence data started at battle init.
btl_scene_variant_rows = 0x801F4AF8;  # data: Scene variant roll threshold rows.
btl_form_tech_table = 0x801F4B1C;  # data: Per-form first and second tech table for transformed party members.
btl_scene_variant = 0x801F4B6C;  # data: Scene variant index used when mapping battle id to scene.
btl_symtbl_ptr = 0x801F4B48;  # data: Pointer to the loaded symtbl (battle id x 8 variants -> scene) table.
btl_scenedat_ptr = 0x801F4B4C;  # data: Pointer to the loaded scenedat records; advanced to the selected scene record.
btl_colour_row_flag = 0x801F4B54;  # data: Set when a party member's battle grid holds the Chrono Cross element (0x7E); turns on the colour-history row.
btl_place_party = 0x801F0BB0;  # function: Places the party in ascending character id: the first PC, then (after the held intro wait) starts a random party intro and loads / places the others one by one
btl_init_party_relief = 0x801F27F8;  # function: first battle-init step, run before the scene select: resets the battle work, copies the party ids and applies the Relief Charm swap unless the battle id is 0x3DD
btlinit_globals_reset = 0x801F2390;  # function: resets the battle globals at battle start, the ring summon colour (0 = none) among them
